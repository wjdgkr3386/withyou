CREATE TABLE users (
   id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '고유 아이디',
   name VARCHAR(50) NOT NULL COMMENT '이름',
   username VARCHAR(50) UNIQUE NOT NULL COMMENT '아이디',
   password VARCHAR(200) NOT NULL COMMENT '비밀번호',
   phone VARCHAR(11) UNIQUE NOT NULL COMMENT '전화번호',
   email VARCHAR(50) UNIQUE COMMENT '이메일'
);
